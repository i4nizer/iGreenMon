# Hotspot
    sudo systemctl disable systemd-resolved.service
    sudo systemctl stop systemd-resolved.service
    sudo systemctl status systemd-resolved.service
    sudo curl -L https://raw.githubusercontent.com/garywill/linux-router/master/lnxrouter -o lnxrouter
    sudo chmod +x lnxrouter
    sudo mv lnxrouter /usr/local/bin/
    sudo lnxrouter --ap wlan0 "iGreenMon" -p "<HOTSPOT_PASSWORD>" --no-virt
    // ap will appear and internet is shared properly

# Persistent Hotspot
    sudo nano /usr/local/bin/hotspot.sh
    sudo chmod +x /usr/local/bin/hotspot.sh
    sudo nano /etc/systemd/system/hotspot.service
    sudo systemctl daemon-reload
    sudo systemctl enable hotspot.service
    sudo systemctl start hotspot.service
    sudo systemctl status hotspot.service
    sudo reboot

# /usr/local/bin/hotspot.sh
    #!/bin/bash
    # Start WiFi Hotspot via lnxrouter

    # Interface for hotspot
    WIFI_IFACE="wlan0"

    # Hotspot SSID and password
    SSID="MyOrangePiHotspot"
    PASS="<HOTSPOT_PASSWORD>"

    # Start AP (no internet sharing needed if ethernet not present)
    lnxrouter --ap "$WIFI_IFACE" "$SSID" -p "$PASS" --no-virt

# /etc/systemd/system/hotspot.service
    [Unit]
    Description=OrangePi Hotspot using lnxrouter
    After=network.target
    Wants=network.target

    [Service]
    Type=simple
    ExecStart=/usr/local/bin/hotspot.sh
    Restart=always
    RestartSec=5
    User=root

    [Install]
    WantedBy=multi-user.target

# nginx
    sudo apt install nginx openssl -y 
    sudo mkdir -p /etc/ssl/igreenmon
    sudo openssl req -x509 -nodes -days 3650 -newkey rsa:4096 \
        -keyout /etc/ssl/igreenmon/igreenmon.key \
        -out /etc/ssl/igreenmon/igreenmon.crt
    sudo nano /etc/nginx/sites-available/igreenmon
    sudo ln -s /etc/nginx/sites-available/igreenmon /etc/nginx/sites-enabled/
    sudo nginx -t
    sudo systemctl restart nginx

# /etc/nginx/sites-available/igreenmon
    server {
        listen 80;
        server_name igreenmon.local;

        # Redirect all HTTP to HTTPS
        return 301 https://$host$request_uri;
    }

    server {
        listen 443 ssl;
        server_name igreenmon.local;

        ssl_certificate /etc/ssl/igreenmon/igreenmon.crt;
        ssl_certificate_key /etc/ssl/igreenmon/igreenmon.key;

        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_prefer_server_ciphers on;

        location / {
            proxy_pass http://127.0.0.1:3000;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection 'upgrade';
            proxy_set_header Host $host;
            proxy_cache_bypass $http_upgrade;
        }
    }
